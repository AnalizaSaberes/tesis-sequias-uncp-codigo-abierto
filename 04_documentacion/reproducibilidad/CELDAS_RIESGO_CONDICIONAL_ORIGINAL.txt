ajustar_marginal_lmom <- function(x, dist_name) {
  x <- as.numeric(x)
  x <- x[is.finite(x) & x > 0]
  lmr <- lmoms(x)
  lmom2par(lmr, type = dist_name)
}

cdf_marginal <- function(x, par_obj) {
  p <- plmomco(x, par_obj)
  pmin(pmax(p, 1e-8), 1 - 1e-8)
}

prob_excedencia_conjunta <- function(d, s, par_d, par_s, theta_gumbel) {
  
  u <- cdf_marginal(d, par_d)
  v <- cdf_marginal(s, par_s)
  
  cop <- gumbelCopula(param = theta_gumbel, dim = 2)
  Cuv <- pCopula(cbind(u, v), cop)
  
  p_joint <- 1 - u - v + Cuv
  p_d     <- 1 - u
  p_s     <- 1 - v
  
  tibble(
    d = d,
    s = s,
    u = u,
    v = v,
    Cuv = Cuv,
    p_joint = p_joint,
    p_d = p_d,
    p_s = p_s
  )
}

riesgo_condicional <- function(d, s, par_d, par_s, theta_gumbel) {
  
  probs <- prob_excedencia_conjunta(d, s, par_d, par_s, theta_gumbel)
  
  probs %>%
    mutate(
      p_s_d = p_joint / p_d,   # P(S > s | D > d)
      p_d_s = p_joint / p_s    # P(D > d | S > s)
    )
}

procesar_cluster_condicional <- function(datos_cluster, cluster_id, dist_d, dist_s, theta_gumbel) {
  
  par_d <- ajustar_marginal_lmom(datos_cluster$duracion, dist_d)
  par_s <- ajustar_marginal_lmom(datos_cluster$severidad, dist_s)
  
  perc_d <- quantile(datos_cluster$duracion, probs = c(0.50, 0.75, 0.90), na.rm = TRUE)
  perc_s <- quantile(datos_cluster$severidad, probs = c(0.50, 0.75, 0.90), na.rm = TRUE)
  
  grid <- expand.grid(
    d = as.numeric(perc_d),
    s = as.numeric(perc_s)
  ) %>%
    as_tibble()
  
  tabla_cond <- grid %>%
    mutate(
      cluster = as.character(cluster_id),
      dist_d = toupper(dist_d),
      dist_s = toupper(dist_s),
      theta = theta_gumbel
    ) %>%
    bind_cols(
      riesgo_condicional(
        d = grid$d,
        s = grid$s,
        par_d = par_d,
        par_s = par_s,
        theta_gumbel = theta_gumbel
      ) %>%
        select(u, v, Cuv, p_joint, p_d, p_s, p_s_d, p_d_s)
    )
  
  list(
    tabla = tabla_cond,
    par_d = par_d,
    par_s = par_s
  )
}

malla_condicional <- function(datos_cluster, par_d, par_s, theta_gumbel, cluster_label,
                              n_d = 60, n_s = 60) {
  
  d_seq <- seq(min(datos_cluster$duracion), quantile(datos_cluster$duracion, 0.95), length.out = n_d)
  s_seq <- seq(min(datos_cluster$severidad), quantile(datos_cluster$severidad, 0.95), length.out = n_s)
  
  grid <- expand.grid(d = d_seq, s = s_seq) %>%
    as_tibble()
  
  cond <- riesgo_condicional(
    d = grid$d,
    s = grid$s,
    par_d = par_d,
    par_s = par_s,
    theta_gumbel = theta_gumbel
  )
  
  bind_cols(grid, cond %>% select(p_s_d, p_d_s)) %>%
    mutate(cluster = cluster_label)
}

extraer_percentiles <- function(datos_cluster) {
  tibble(
    tipo = c("P50", "P75", "P90"),
    d = as.numeric(quantile(datos_cluster$duracion, probs = c(0.50, 0.75, 0.90), na.rm = TRUE)),
    s = as.numeric(quantile(datos_cluster$severidad, probs = c(0.50, 0.75, 0.90), na.rm = TRUE))
  )
}

plot_condicional_individual <- function(malla_df, datos_cluster, cluster_label,
                                        tipo = c("p_s_d", "p_d_s")) {
  
  tipo <- match.arg(tipo)
  percs <- extraer_percentiles(datos_cluster)
  
  if (tipo == "p_s_d") {
    zvar <- "p_s_d"
    titulo <- paste0(cluster_label, " | Riesgo condicional: P(S > s | D > d)")
    subtitulo <- "Probabilidad de exceder severidad dado que la duraciÃ³n excede el umbral"
    legend_lab <- expression(P(S>s~"|"~D>d))
  } else {
    zvar <- "p_d_s"
    titulo <- paste0(cluster_label, " | Riesgo condicional: P(D > d | S > s)")
    subtitulo <- "Probabilidad de exceder duraciÃ³n dado que la severidad excede el umbral"
    legend_lab <- expression(P(D>d~"|"~S>s))
  }
  
  brks <- pretty(range(malla_df[[zvar]], na.rm = TRUE), n = 7)
  
  ggplot(malla_df, aes(x = d, y = s)) +
    geom_raster(aes(fill = .data[[zvar]]), interpolate = TRUE) +
    geom_contour(
      aes(z = .data[[zvar]], color = after_stat(level)),
      breaks = brks,
      linewidth = 0.55,
      show.legend = FALSE
    ) +
    metR::geom_text_contour(
      aes(z = .data[[zvar]]),
      breaks = brks,
      stroke = 0.15,
      size = 3.2,
      check_overlap = TRUE
    ) +
    geom_vline(
      data = percs,
      aes(xintercept = d),
      linetype = "dashed",
      linewidth = 0.45,
      color = "#4d4d4d"
    ) +
    geom_hline(
      data = percs,
      aes(yintercept = s),
      linetype = "dashed",
      linewidth = 0.45,
      color = "#4d4d4d"
    ) +
    geom_text(
      data = percs,
      aes(x = d, y = min(malla_df$s, na.rm = TRUE), label = tipo),
      inherit.aes = FALSE,
      vjust = 1.8,
      size = 3.3,
      color = "#3a3a3a",
      fontface = "bold"
    ) +
    geom_text(
      data = percs,
      aes(x = min(malla_df$d, na.rm = TRUE), y = s, label = tipo),
      inherit.aes = FALSE,
      hjust = -0.15,
      size = 3.3,
      color = "#3a3a3a",
      fontface = "bold"
    ) +
    scale_fill_gradientn(
      colours = pal_sequia,
      name = legend_lab,
      limits = c(0, 1)
    ) +
    labs(
      title = titulo,
      subtitle = subtitulo,
      x = "DuraciÃ³n (meses)",
      y = "Severidad"
    ) +
    theme_q1_single()
}
