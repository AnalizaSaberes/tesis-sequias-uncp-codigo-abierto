ajustar_marginal_lmom <- function(x, dist_name) {
  x <- as.numeric(x)
  x <- x[is.finite(x) & x > 0]
  lmr <- lmoms(x)
  lmom2par(lmr, type = dist_name)
}

# Inversa aproximada por interpolaciÃ³n numÃ©rica
inversa_marginal_lmom <- function(u, par_obj, xmax = NULL, n_grid = 5000) {
  u <- pmin(pmax(u, 1e-8), 1 - 1e-8)

  if (is.null(xmax)) {
    xmax <- max(100, max(par_obj$para, na.rm = TRUE) * 10)
  }

  xgrid <- seq(1e-6, xmax, length.out = n_grid)
  pgrid <- plmomco(xgrid, par_obj)

  ok <- is.finite(pgrid) & !is.na(pgrid)
  xgrid <- xgrid[ok]
  pgrid <- pgrid[ok]

  # asegurar monotonicidad
  ord <- order(pgrid)
  pgrid <- pgrid[ord]
  xgrid <- xgrid[ord]

  # quitar duplicados
  keep <- !duplicated(pgrid)
  pgrid <- pgrid[keep]
  xgrid <- xgrid[keep]

  approx(x = pgrid, y = xgrid, xout = u, rule = 2)$y
}

simular_cluster_mc <- function(datos_cluster, cluster_id, dist_d, dist_s, theta_gumbel,
                               n_sim = 10000, semilla = 123) {

  set.seed(semilla)

  # Ajustar marginales
  par_d <- ajustar_marginal_lmom(datos_cluster$duracion, dist_d)
  par_s <- ajustar_marginal_lmom(datos_cluster$severidad, dist_s)

  # Definir mÃ¡ximos de interpolaciÃ³n
  xmax_d <- max(datos_cluster$duracion, na.rm = TRUE) * 3
  xmax_s <- max(datos_cluster$severidad, na.rm = TRUE) * 3

  # Simular de la cÃ³pula
  cop <- gumbelCopula(param = theta_gumbel, dim = 2)
  uv <- rCopula(n_sim, cop)

  u <- uv[, 1]
  v <- uv[, 2]

  # Transformar al espacio real
  d_sim <- inversa_marginal_lmom(u, par_d, xmax = xmax_d)
  s_sim <- inversa_marginal_lmom(v, par_s, xmax = xmax_s)

  tibble(
    cluster = cluster_id,
    u = u,
    v = v,
    duracion_sim = d_sim,
    severidad_sim = s_sim
  )
}

resumen_predictivo <- function(sim_df, datos_cluster) {

  p50_d <- quantile(datos_cluster$duracion, 0.50, na.rm = TRUE)
  p75_d <- quantile(datos_cluster$duracion, 0.75, na.rm = TRUE)
  p90_d <- quantile(datos_cluster$duracion, 0.90, na.rm = TRUE)

  p50_s <- quantile(datos_cluster$severidad, 0.50, na.rm = TRUE)
  p75_s <- quantile(datos_cluster$severidad, 0.75, na.rm = TRUE)
  p90_s <- quantile(datos_cluster$severidad, 0.90, na.rm = TRUE)

  tibble(
    n_sim = nrow(sim_df),

    duracion_media = mean(sim_df$duracion_sim, na.rm = TRUE),
    duracion_mediana = median(sim_df$duracion_sim, na.rm = TRUE),
    duracion_p05 = quantile(sim_df$duracion_sim, 0.05, na.rm = TRUE),
    duracion_p95 = quantile(sim_df$duracion_sim, 0.95, na.rm = TRUE),

    severidad_media = mean(sim_df$severidad_sim, na.rm = TRUE),
    severidad_mediana = median(sim_df$severidad_sim, na.rm = TRUE),
    severidad_p05 = quantile(sim_df$severidad_sim, 0.05, na.rm = TRUE),
    severidad_p95 = quantile(sim_df$severidad_sim, 0.95, na.rm = TRUE),

    prob_d_gt_p50 = mean(sim_df$duracion_sim > p50_d, na.rm = TRUE),
    prob_d_gt_p75 = mean(sim_df$duracion_sim > p75_d, na.rm = TRUE),
    prob_d_gt_p90 = mean(sim_df$duracion_sim > p90_d, na.rm = TRUE),

    prob_s_gt_p50 = mean(sim_df$severidad_sim > p50_s, na.rm = TRUE),
    prob_s_gt_p75 = mean(sim_df$severidad_sim > p75_s, na.rm = TRUE),
    prob_s_gt_p90 = mean(sim_df$severidad_sim > p90_s, na.rm = TRUE),

    prob_joint_p75 = mean(sim_df$duracion_sim > p75_d & sim_df$severidad_sim > p75_s, na.rm = TRUE),
    prob_joint_p90 = mean(sim_df$duracion_sim > p90_d & sim_df$severidad_sim > p90_s, na.rm = TRUE)
  )
}

clasificar_escenario <- function(sim_df, datos_cluster) {

  p75_d <- quantile(datos_cluster$duracion, 0.75, na.rm = TRUE)
  p90_d <- quantile(datos_cluster$duracion, 0.90, na.rm = TRUE)

  p75_s <- quantile(datos_cluster$severidad, 0.75, na.rm = TRUE)
  p90_s <- quantile(datos_cluster$severidad, 0.90, na.rm = TRUE)

  sim_df %>%
    mutate(
      escenario = case_when(
        duracion_sim > p90_d & severidad_sim > p90_s ~ "Extremo",
        duracion_sim > p75_d & severidad_sim > p75_s ~ "Severo",
        TRUE ~ "TÃ­pico"
      )
    )
}
