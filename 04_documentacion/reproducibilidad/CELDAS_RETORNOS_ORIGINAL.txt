prob_and_or <- function(d, s, par_d, par_s, theta_gumbel) {
  
  u <- cdf_marginal(d, par_d)
  v <- cdf_marginal(s, par_s)
  
  cop <- gumbelCopula(param = theta_gumbel, dim = 2)
  Cuv <- pCopula(cbind(u, v), cop)
  
  p_and <- 1 - u - v + Cuv
  p_or  <- 1 - Cuv
  
  tibble(
    d = d,
    s = s,
    u = u,
    v = v,
    Cuv = Cuv,
    p_and = p_and,
    p_or = p_or
  )
}

retorno_and_or <- function(d, s, par_d, par_s, theta_gumbel, lambda_eventos) {
  
  probs <- prob_and_or(d, s, par_d, par_s, theta_gumbel)
  
  probs %>%
    mutate(
      T_and = 1 / (lambda_eventos * p_and),
      T_or  = 1 / (lambda_eventos * p_or)
    )
}

ajustar_marginal_lmom <- function(x, dist_name) {

}

prob_and_or <- function(d, s, par_d, par_s, theta_gumbel) {
  
  u <- cdf_marginal(d, par_d)
  v <- cdf_marginal(s, par_s)
  
  cop <- gumbelCopula(param = theta_gumbel, dim = 2)

retorno_and_or <- function(d, s, par_d, par_s, theta_gumbel, lambda_eventos) {
  
  probs <- prob_and_or(d, s, par_d, par_s, theta_gumbel)

procesar_cluster_retorno <- function(datos_cluster, cluster_id, dist_d, dist_s, theta_gumbel, N_anios = 54) {
  
  lambda_eventos <- nrow(datos_cluster) / N_anios
  
  par_d <- ajustar_marginal_lmom(datos_cluster$duracion, dist_d)
  par_s <- ajustar_marginal_lmom(datos_cluster$severidad, dist_s)
  
  perc_d <- quantile(datos_cluster$duracion, probs = c(0.50, 0.75, 0.90), na.rm = TRUE)
  perc_s <- quantile(datos_cluster$severidad, probs = c(0.50, 0.75, 0.90), na.rm = TRUE)
  
  grid <- expand.grid(
    d = as.numeric(perc_d),
    s = as.numeric(perc_s)
  ) %>%
    as_tibble()
  
  tabla_ret <- grid %>%
    mutate(
      cluster = as.character(cluster_id),
      dist_d = toupper(dist_d),
      dist_s = toupper(dist_s),
      theta = theta_gumbel,
      lambda = lambda_eventos
    ) %>%
    bind_cols(
      retorno_and_or(
        d = grid$d,
        s = grid$s,
        par_d = par_d,
        par_s = par_s,
        theta_gumbel = theta_gumbel,
        lambda_eventos = lambda_eventos
      ) %>%
        select(u, v, Cuv, p_and, p_or, T_and, T_or)
    )
  
  list(
    tabla = tabla_ret,
    par_d = par_d,
    par_s = par_s,
    lambda = lambda_eventos
  )
}

malla_retornos <- function(datos_cluster, par_d, par_s, theta_gumbel, lambda_eventos,
                           cluster_label, n_d = 60, n_s = 60) {
  
  d_seq <- seq(min(datos_cluster$duracion), quantile(datos_cluster$duracion, 0.95), length.out = n_d)
  s_seq <- seq(min(datos_cluster$severidad), quantile(datos_cluster$severidad, 0.95), length.out = n_s)
  
  grid <- expand.grid(d = d_seq, s = s_seq) %>%
    as_tibble()
  
  ret <- retorno_and_or(
    d = grid$d,
    s = grid$s,
    par_d = par_d,
    par_s = par_s,
    theta_gumbel = theta_gumbel,
    lambda_eventos = lambda_eventos
  )
  
  bind_cols(grid, ret %>% select(T_and, T_or)) %>%
    mutate(cluster = cluster_label)
}

extraer_percentiles <- function(datos_cluster) {
  tibble(
    tipo = c("P50", "P75", "P90"),
    d = as.numeric(quantile(datos_cluster$duracion, probs = c(0.50, 0.75, 0.90), na.rm = TRUE)),
    s = as.numeric(quantile(datos_cluster$severidad, probs = c(0.50, 0.75, 0.90), na.rm = TRUE))
  )
}

# tipo = "AND" o "OR"
# ---------------------------------------------------------
plot_retorno_individual <- function(malla_df, datos_cluster, cluster_label, tipo = c("AND", "OR")) {
  
  tipo <- match.arg(tipo)
  percs <- extraer_percentiles(datos_cluster)
  
  if (tipo == "AND") {
    zvar <- "T_and"
    titulo <- paste0(cluster_label, " | Periodo de retorno conjunto tipo AND")
    subtitulo <- "Escala nSPEI-3 con cÃ³pula de Gumbel"
    legend_lab <- expression(T[AND]~"(aÃ±os)")
  } else {
    zvar <- "T_or"
    titulo <- paste0(cluster_label, " | Periodo de retorno conjunto tipo OR")
    subtitulo <- "Escala nSPEI-3 con cÃ³pula de Gumbel"
    legend_lab <- expression(T[OR]~"(aÃ±os)")
  }
  
  # niveles de contorno
  brks <- pretty(range(malla_df[[zvar]], na.rm = TRUE), n = 7)
  
  ggplot(malla_df, aes(x = d, y = s)) +
    geom_raster(aes(fill = .data[[zvar]]), interpolate = TRUE) +
    geom_contour(
      aes(z = .data[[zvar]], color = after_stat(level)),
      breaks = brks,
      linewidth = 0.55,
      show.legend = FALSE
    ) +
    metR::geom_text_contour(
      aes(z = .data[[zvar]]),
      breaks = brks,
      stroke = 0.15,
      size = 3.2,
      check_overlap = TRUE
    ) +
    geom_vline(
      data = percs,
      aes(xintercept = d),
      linetype = "dashed",
      linewidth = 0.45,
      color = "#4d4d4d"
    ) +
    geom_hline(
      data = percs,
      aes(yintercept = s),
      linetype = "dashed",
      linewidth = 0.45,
      color = "#4d4d4d"
    ) +
    geom_text(
      data = percs,
      aes(x = d, y = min(malla_df$s, na.rm = TRUE), label = tipo),
      inherit.aes = FALSE,
      vjust = 1.8,
      size = 3.3,
      color = "#3a3a3a",
      fontface = "bold"
    ) +
    geom_text(
      data = percs,
      aes(x = min(malla_df$d, na.rm = TRUE), y = s, label = tipo),
      inherit.aes = FALSE,
      hjust = -0.15,
      size = 3.3,
      color = "#3a3a3a",
      fontface = "bold"
    ) +
    scale_fill_gradientn(
      colours = pal_sequia,
      name = legend_lab
    ) +
    labs(
      title = titulo,
      subtitle = subtitulo,
      x = "DuraciÃ³n (meses)",
      y = "Severidad"
    ) +
    theme_q1_single()
}
